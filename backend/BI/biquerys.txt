CREATE TABLE IF NOT EXISTS public.bi_purchase
(
    id bigint,
    version integer,
    invoiceno character varying(50) COLLATE pg_catalog."default",
    entrydate timestamp without time zone,
    invoicedate timestamp without time zone,
    itbuscode character varying(100) COLLATE pg_catalog."default",
    barcode character varying(100) COLLATE pg_catalog."default",
    serialno character varying(100) COLLATE pg_catalog."default",
    supplier character varying(250) COLLATE pg_catalog."default",
    buyingqty numeric(20,6),
    buyingprice numeric(20,6),
    amount numeric,
    purchasetax numeric(20,6),
    purchasediscount numeric(20,6),
    product character varying(100) COLLATE pg_catalog."default",
    productgroup character varying(100) COLLATE pg_catalog."default",
    pgh_1 character varying(100) COLLATE pg_catalog."default",
    pgh_2 character varying(100) COLLATE pg_catalog."default",
    pgh_3 character varying(100) COLLATE pg_catalog."default",
    pgh_4 character varying(100) COLLATE pg_catalog."default",
    pgh_5 character varying(100) COLLATE pg_catalog."default",
    brand character varying(100) COLLATE pg_catalog."default",
    gln character varying(100) COLLATE pg_catalog."default",
    size character varying(100) COLLATE pg_catalog."default",
    style character varying(100) COLLATE pg_catalog."default"
);
